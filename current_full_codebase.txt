import React, { useRef } from 'react';
import { TouchableWithoutFeedback, Animated, ViewStyle, StyleProp } from 'react-native';

interface M3PressableProps {
  onPress?: () => void;
  style?: StyleProp<ViewStyle>;
  children: React.ReactNode;
  pressedScale?: number;
}

export const M3Pressable: React.FC<M3PressableProps> = ({
  onPress,
  style,
  children,
  pressedScale = 0.94,
}) => {
  const scaleAnim = useRef(new Animated.Value(1)).current;

  const handlePressIn = () => {
    Animated.spring(scaleAnim, {
      toValue: pressedScale,
      tension: 180,
      friction: 12,
      useNativeDriver: true,
    }).start();
  };

  const handlePressOut = () => {
    Animated.spring(scaleAnim, {
      toValue: 1,
      tension: 140,
      friction: 10,
      useNativeDriver: true,
    }).start();
  };

  return (
    <TouchableWithoutFeedback
      onPress={onPress}
      onPressIn={handlePressIn}
      onPressOut={handlePressOut}
    >
      <Animated.View style={[style, { transform: [{ scale: scaleAnim }] }]}>
        {children}
      </Animated.View>
    </TouchableWithoutFeedback>
  );
};

import React, { useEffect, useRef, useState, useMemo } from 'react';
import {
  View,
  StyleSheet,
  PanResponder,
  Animated,
  Easing,
  GestureResponderEvent,
  PanResponderGestureState,
} from 'react-native';
import Svg, { Path, Circle } from 'react-native-svg';
import * as Haptics from 'expo-haptics';

interface WavySliderExpressiveProps {
  progressFraction: number; // 0.0 to 1.0
  isPlaying: boolean;
  onSeekChange: (fraction: number) => void;
  onSeekCommit: (fraction: number) => void;
  activeTrackColor: string;
  inactiveTrackColor: string;
  thumbColor: string;
  trackWidth: number;
}

const WAVELENGTH = 30; // 30.dp from Kotlin spec
const WAVE_AMPLITUDE = 3.8;
const TRACK_HEIGHT = 28;
const THUMB_RADIUS = 7;

export const WavySliderExpressive: React.FC<WavySliderExpressiveProps> = ({
  progressFraction,
  isPlaying,
  onSeekChange,
  onSeekCommit,
  activeTrackColor,
  inactiveTrackColor,
  thumbColor,
  trackWidth,
}) => {
  const [isDragging, setIsDragging] = useState(false);
  const [dragFraction, setDragFraction] = useState<number | null>(null);
  const lastHapticStep = useRef(-1);

  // Phase animation for continuous sine wave undulation
  const phaseAnim = useRef(new Animated.Value(0)).current;
  // Amplitude animation: smoothly transitions between wavy (playing) and flat (paused)
  const amplitudeAnim = useRef(new Animated.Value(isPlaying ? 1 : 0)).current;
  const [phaseValue, setPhaseValue] = useState(0);
  const [ampValue, setAmpValue] = useState(isPlaying ? 1 : 0);

  useEffect(() => {
    const idPhase = phaseAnim.addListener(({ value }) => setPhaseValue(value));
    const idAmp = amplitudeAnim.addListener(({ value }) => setAmpValue(value));
    return () => {
      phaseAnim.removeListener(idPhase);
      amplitudeAnim.removeListener(idAmp);
    };
  }, [phaseAnim, amplitudeAnim]);

  // Undulation animation loop
  useEffect(() => {
    let loopAnimation: Animated.CompositeAnimation | null = null;
    if (isPlaying) {
      loopAnimation = Animated.loop(
        Animated.timing(phaseAnim, {
          toValue: 1,
          duration: 1200,
          easing: Easing.linear,
          useNativeDriver: false,
        })
      );
      loopAnimation.start();
    } else {
      phaseAnim.stopAnimation();
    }
    return () => loopAnimation?.stop();
  }, [isPlaying, phaseAnim]);

  // Smooth flat <-> wave morphing
  useEffect(() => {
    Animated.timing(amplitudeAnim, {
      toValue: isPlaying ? 1 : 0,
      duration: 350,
      easing: Easing.bezier(0.4, 0.0, 0.2, 1.0),
      useNativeDriver: false,
    }).start();
  }, [isPlaying, amplitudeAnim]);

  const effectiveFraction = isDragging && dragFraction !== null ? dragFraction : progressFraction;
  const clampedFraction = Math.min(Math.max(0, effectiveFraction), 1);
  const currentHeadX = clampedFraction * trackWidth;

  // 20-step quantized haptic feedback (every 5% step)
  const triggerQuantizedHaptic = (fraction: number) => {
    const step = Math.floor(Math.min(Math.max(0, fraction), 1) * 20);
    if (step !== lastHapticStep.current) {
      lastHapticStep.current = step;
      try {
        Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
      } catch {
        // Haptics fallback on web or devices lacking haptic motors
      }
    }
  };

  const panResponder = useMemo(
    () =>
      PanResponder.create({
        onStartShouldSetPanResponder: () => true,
        onMoveShouldSetPanResponder: () => true,
        onPanResponderGrant: (evt: GestureResponderEvent) => {
          setIsDragging(true);
          const { locationX } = evt.nativeEvent;
          const frac = Math.min(Math.max(0, locationX / trackWidth), 1);
          setDragFraction(frac);
          triggerQuantizedHaptic(frac);
          onSeekChange(frac);
        },
        onPanResponderMove: (_: GestureResponderEvent, gesture: PanResponderGestureState) => {
          const newX = Math.min(Math.max(0, currentHeadX + gesture.dx), trackWidth);
          const frac = newX / trackWidth;
          setDragFraction(frac);
          triggerQuantizedHaptic(frac);
          onSeekChange(frac);
        },
        onPanResponderRelease: () => {
          setIsDragging(false);
          const finalFrac = dragFraction ?? clampedFraction;
          onSeekCommit(finalFrac);
          setDragFraction(null);
          lastHapticStep.current = -1;
        },
        onPanResponderTerminate: () => {
          setIsDragging(false);
          setDragFraction(null);
          lastHapticStep.current = -1;
        },
      }),
    [trackWidth, currentHeadX, dragFraction, clampedFraction, onSeekChange, onSeekCommit]
  );

  // Generate SVG path for the active wavy track up to the thumb head
  const centerY = TRACK_HEIGHT / 2;
  const activePathData = useMemo(() => {
    if (currentHeadX <= 0) return '';
    let d = `M 0 ${centerY}`;
    const stepPx = 3;
    const currentAmp = WAVE_AMPLITUDE * ampValue;
    const phaseOffset = phaseValue * 2 * Math.PI;

    for (let x = 0; x <= currentHeadX; x += stepPx) {
      const angle = (x / WAVELENGTH) * 2 * Math.PI - phaseOffset;
      const y = centerY + Math.sin(angle) * currentAmp;
      d += ` L ${x.toFixed(1)} ${y.toFixed(1)}`;
    }
    d += ` L ${currentHeadX.toFixed(1)} ${centerY}`;
    return d;
  }, [currentHeadX, centerY, ampValue, phaseValue]);

  // Inactive straight track extending from the thumb head to the end
  const inactivePathData = useMemo(() => {
    return `M ${Math.max(0, currentHeadX).toFixed(1)} ${centerY} L ${trackWidth.toFixed(1)} ${centerY}`;
  }, [currentHeadX, trackWidth, centerY]);

  return (
    <View style={styles.container} {...panResponder.panHandlers}>
      <Svg width={trackWidth} height={TRACK_HEIGHT}>
        {/* Inactive tail */}
        <Path
          d={inactivePathData}
          stroke={inactiveTrackColor}
          strokeWidth={4}
          strokeLinecap="round"
        />

        {/* Active oscillating sine wave */}
        {activePathData ? (
          <Path
            d={activePathData}
            stroke={activeTrackColor}
            strokeWidth={4.5}
            strokeLinecap="round"
            strokeLinejoin="round"
            fill="none"
          />
        ) : null}

        {/* Scrubber thumb circle */}
        <Circle
          cx={Math.max(THUMB_RADIUS, Math.min(trackWidth - THUMB_RADIUS, currentHeadX))}
          cy={centerY}
          r={isDragging ? THUMB_RADIUS + 2.5 : THUMB_RADIUS}
          fill={thumbColor}
        />
      </Svg>
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    height: TRACK_HEIGHT,
    justifyContent: 'center',
    paddingVertical: 2,
  },
});

  const insets = useSafeAreaInsets();
  const slideAnim = useRef(new Animated.Value(SCREEN_HEIGHT)).current;
  const fadeAnim = useRef(new Animated.Value(0)).current;

  // Accordion Expand/Collapse States
  const [isThemePickerOpen, setIsThemePickerOpen] = useState(false);
  const [isNetworkOpen, setIsNetworkOpen] = useState(false);
  const [isCookieSyncOpen, setIsCookieSyncOpen] = useState(false);
  const [isAdminOpen, setIsAdminOpen] = useState(false);

  // Functional Feature States
  const [networkMode, setNetworkMode] = useState<NetworkMode>('cloud');
  const [isWakingHost, setIsWakingHost] = useState(false);
  const [isCleaningStorage, setIsCleaningStorage] = useState(false);
  const [isCheckingUpdate, setIsCheckingUpdate] = useState(false);
  const [cookieText, setCookieText] = useState('');
  const [isSyncingCookies, setIsSyncingCookies] = useState(false);
  const [cookieFeedback, setCookieFeedback] = useState<{

    message: string;

    type: 'idle',
    message: '',
  });

  // Admin Verification State
  const [adminPassword, setAdminPassword] = useState('');
  const [adminKey, setAdminKey] = useState('');
  const [isAdminAuthenticated, setIsAdminAuthenticated] = useState(false);
  const [adminError, setAdminError] = useState<string | null>(null);

  // In-sheet Toast
  const [toastMessage, setToastMessage] = useState<string | null>(null);

  const showToast = (msg: string) => {
    setToastMessage(msg);
    setTimeout(() => setToastMessage(null), 3000);
  };

  useEffect(() => {
    if (visible) {
      setNetworkMode(getSavedNetworkMode());
      Animated.parallel([
        Animated.timing(fadeAnim, { toValue: 1, duration: 220, useNativeDriver: true }),
        Animated.spring(slideAnim, { toValue: 0, tension: 65, friction: 9, useNativeDriver: true }),
      ]).start();
    } else {
      Animated.parallel([
        Animated.timing(fadeAnim, { toValue: 0, duration: 180, useNativeDriver: true }),
        Animated.timing(slideAnim, { toValue: SCREEN_HEIGHT, duration: 220, useNativeDriver: true }),
      ]).start();
    }
  }, [visible, fadeAnim, slideAnim]);

  const handleClose = () => {
    try {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
    } catch {}
    Animated.parallel([

        if (gesture.dy > 0) {

      },
      onPanResponderRelease: (_, gesture) => {
        if (gesture.dy > 120 || gesture.vy > 0.6) {
          handleClose();
        } else {
          Animated.spring(slideAnim, { toValue: 0, tension: 80, friction: 9, useNativeDriver: true }).start();
        }
      },
    })
  ).current;

  // 1. Network Strategy Selection
  const handleSelectNetworkMode = async (mode: NetworkMode) => {
    try {
      Haptics.selectionAsync();

    await setSavedNetworkMode(mode);
    setNetworkMode(mode);
    showToast(`Route: ${NETWORK_MODES[mode].label}`);
  };

  // 2. Wake Up Render Host

      showToast(`Cold start pending (${result.timeMs}ms).`);

    if (isCleaningStorage) return;

      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);

        showToast(`Storage optimal (${storageUsage}). 0 orphans.`);
      }
    } catch {
      showToast('Cleanup failed.');
    } finally {
      setIsCleaningStorage(false);
    }
  };

    if (isCheckingUpdate) return;
    try {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);

const safeDecode = (str: string | null): string => {
  if (!str) return '';
  try {
    return decodeURIComponent(str);
  } catch {
    return str;
  }
};

export const parseDurationToSeconds = (input: any): number => {
  if (!input) return 0;
  if (typeof input === 'number') {
    return input > 1000 ? Math.floor(input / 1000) : Math.floor(input);
  }
  if (typeof input === 'string') {
    if (input.includes(':')) {
      const parts = input.split(':').map((p) => parseInt(p, 10) || 0);
      if (parts.length === 3) return parts[0] * 3600 + parts[1] * 60 + parts[2];
      if (parts.length === 2) return parts[0] * 60 + parts[1];
    }
    const parsed = parseFloat(input);
    if (!isNaN(parsed) && parsed > 0) {
      return parsed > 1000 ? Math.floor(parsed / 1000) : Math.floor(parsed);
    }
  }
  return 0;
};

export const formatDuration = (seconds: number | string | undefined | null): string => {
  if (seconds === undefined || seconds === null) return '--:--';
  if (typeof seconds === 'string' && seconds.includes(':')) return seconds.trim();

  const total = typeof seconds === 'string' ? parseFloat(seconds) : seconds;
  if (isNaN(total) || total <= 0) return '--:--';

  const secsTotal = total > 1000 ? Math.floor(total / 1000) : Math.floor(total);
  const mins = Math.floor(secsTotal / 60);
  const secs = secsTotal % 60;
  return `${mins}:${secs.toString().padStart(2, '0')}`;
};

// Measures the real duration directly from the downloaded audio file
const detectLocalAudioDuration = async (fileUri: string): Promise<number> => {
  return new Promise((resolve) => {
    let tempPlayer: any = null;
    let timeout: any = null;

    const cleanup = () => {
      if (timeout) clearTimeout(timeout);
      if (tempPlayer) {
        try {
          if (typeof tempPlayer.remove === 'function') tempPlayer.remove();
          else if (typeof tempPlayer.release === 'function') tempPlayer.release();
        } catch {}
        tempPlayer = null;
      }
    };

    timeout = setTimeout(() => {
      cleanup();
      resolve(0);
    }, 1200);

    try {
      tempPlayer = createAudioPlayer({ uri: fileUri });
      if (!tempPlayer) {
        cleanup();
        resolve(0);
        return;
      }

      const immediateDur = typeof tempPlayer.duration === 'number' ? tempPlayer.duration : 0;
      if (immediateDur > 0) {
        cleanup();
        resolve(immediateDur > 1000 ? Math.floor(immediateDur / 1000) : Math.floor(immediateDur));
        return;
      }

      if (typeof tempPlayer.addListener === 'function') {
        tempPlayer.addListener('playbackStatusUpdate', (status: any) => {
          const sDur = status?.duration ?? tempPlayer?.duration ?? 0;
          if (sDur > 0) {
            cleanup();
            resolve(sDur > 1000 ? Math.floor(sDur / 1000) : Math.floor(sDur));
          }
        });
      }
    } catch {
      cleanup();
      resolve(0);
    }
  });
};

export const pingRenderHealth = async (): Promise<{ ok: boolean; timeMs: number }> => {
  const renderUrl = getRenderUrl();
  const startTime = Date.now();

  try {
    const res = await fetch(`${renderUrl}/health`, { method: 'GET' });
    const timeMs = Date.now() - startTime;
    if (!res.ok) return { ok: false, timeMs };
    const data = await res.json();
    return { ok: data?.status === 'ok', timeMs };
  } catch {
    const timeMs = Date.now() - startTime;
    return { ok: false, timeMs };
  }
};

// Lightweight info API (titles, channel, thumbnail only)
export const fetchVideoInfo = async (url: string, signal?: AbortSignal): Promise<VideoInfo> => {
  const { baseUrl, networkMode } = getApiConfig();

  const res = await fetch(`${baseUrl}/converter/info`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ url, networkMode }),
    signal,
  });

  const contentType = res.headers.get('content-type') || '';
  if (!contentType.includes('application/json')) {
    throw new Error(`Received non-JSON response from backend (${baseUrl})`);
  }

  if (!res.ok) {
    const errData = await res.json().catch(() => null);
    throw new Error(errData?.message || `Failed to fetch preview: ${res.statusText}`);
  }

  const data = await res.json();
  const durSec = parseDurationToSeconds(data.duration || data.durationFormatted);

  return {
    ...data,
    duration: durSec,
    durationFormatted: durSec > 0 ? formatDuration(durSec) : '--:--',
  };
};

// Full audio converter: extracts stream and sets real duration upon completion
export const downloadAudioStream = async (
  url: string,
  previewFallback?: VideoInfo | null,
  onProgress?: (progress: ProgressCallbackData) => void,
  maxRetries = 3,
  retryDelayMs = 2000,
  signal?: AbortSignal
): Promise<Track> => {
  let attempt = 0;
  const { baseUrl, networkMode } = getApiConfig();

  while (attempt < maxRetries) {
    if (signal?.aborted) {
      throw new Error('Download aborted by user');
    }

    attempt++;

    try {
      if (attempt > 1 && onProgress) {
        onProgress({
          percent: 0,
          receivedBytes: 0,
          totalBytes: 0,
          statusText: `Retrying download (Attempt ${attempt}/${maxRetries})...`,
        });
      }

      onProgress?.({
        percent: 0,
        receivedBytes: 0,
        totalBytes: 0,
        statusText: 'Connecting to audio stream...',
      });

      const res = await fetch(`${baseUrl}/converter/download`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          url,
          networkMode,
          id: previewFallback?.id,
          title: previewFallback?.title,
          channel: previewFallback?.channel,
          thumbnail: previewFallback?.thumbnail,
        }),
        signal,
      });

      if (!res.ok) {
        throw new Error(`Server returned HTTP ${res.status}: ${res.statusText}`);
      }

      // Check for headers that the download endpoint might return
      const rawTitle = res.headers.get('X-Audio-Title');
      const rawArtist = res.headers.get('X-Audio-Artist');
      const rawThumb = res.headers.get('X-Audio-Thumbnail');
      const rawDuration =
        res.headers.get('X-Audio-Duration') ||
        res.headers.get('x-duration') ||
        res.headers.get('x-audio-duration');

      const title = previewFallback?.title || safeDecode(rawTitle) || 'Unknown Title';
      const artist = previewFallback?.channel || safeDecode(rawArtist) || 'Unknown Artist';
      const thumbnailUrl = previewFallback?.thumbnail || rawThumb || '';
      const id = previewFallback?.id || res.headers.get('X-Audio-Id') || `track_${Date.now()}`;
      const targetFilePath = getTrackFilePath(id);

      onProgress?.({
        percent: 45,
        receivedBytes: 0,
        totalBytes: 0,
        statusText: 'Writing track to device storage...',
      });

      const arrayBuffer = await res.arrayBuffer();
      if (signal?.aborted) {
        throw new Error('Download aborted by user');
      }

      const uint8Array = new Uint8Array(arrayBuffer);

      let binaryString = '';
      const chunkSize = 8192;
      for (let i = 0; i < uint8Array.length; i += chunkSize) {
        binaryString += String.fromCharCode.apply(
          null,
          Array.from(uint8Array.subarray(i, i + chunkSize))
        );
      }
      const base64Data = btoa(binaryString);

      await FileSystem.writeAsStringAsync(targetFilePath, base64Data, {
        encoding: FileSystem.EncodingType.Base64,
      });

      onProgress?.({
        percent: 90,
        receivedBytes: uint8Array.length,
        totalBytes: uint8Array.length,
        statusText: 'Analyzing duration...',
      });

      // 1. Check if the download API provided the duration in headers
      let durationSeconds = parseDurationToSeconds(rawDuration);

      // 2. If not in headers, measure the saved MP3 file directly
      if (durationSeconds <= 0) {
        durationSeconds = await detectLocalAudioDuration(targetFilePath);
      }

      const formatted = formatDuration(durationSeconds);

      onProgress?.({
        percent: 100,
        receivedBytes: uint8Array.length,
        totalBytes: uint8Array.length,
        statusText: 'Saved offline!',
      });

      return {
        id,
        title,
        artist,
        duration: durationSeconds,
        durationFormatted: formatted,
        thumbnailUrl,
        fileUri: targetFilePath,
        createdAt: Date.now(),
        isFavorite: false,
      };
    } catch (err: unknown) {
      if (signal?.aborted) {
        throw new Error('Download aborted by user');
      }

      if (attempt >= maxRetries) {
        const msg = err instanceof Error ? err.message : 'Network error';
        throw new Error(`Download failed after ${maxRetries} attempts: ${msg}`);
      }

      onProgress?.({
        percent: 0,
        receivedBytes: 0,
        totalBytes: 0,
        statusText: `Connection dropped. Retrying in ${retryDelayMs / 1000}s...`,
      });

      await sleep(retryDelayMs);
    }
  }

  throw new Error('Download terminated unexpectedly');
};

export const syncCookiesToBackend = async (
  cookieText: string
): Promise<{ success: boolean; message: string; byteCount: number }> => {
  const { baseUrl } = getApiConfig();

  const res = await fetch(`${baseUrl}/converter/cookies`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ cookies: cookieText.trim() }),
  });

  if (!res.ok) {
    const errorData = await res.json().catch(() => null);
    throw new Error(errorData?.message || 'Failed to update cookies on server.');
  }

  return await res.json();
};

import React, { useState, useEffect, useCallback } from 'react';
import {
  View,
  StyleSheet,

  Platform,
} from 'react-native';
import { SafeAreaProvider, useSafeAreaInsets } from 'react-native-safe-area-context';

import { SearchScreen } from './src/screens/search/SearchScreen';
import { LibraryScreen } from './src/screens/library/LibraryScreen';
import { useAudioPlayer } from './src/hooks/useAudioPlayer';
import {
  getAllTracks,
  saveTrack,
  deleteTrack,
  toggleTrackFavorite,
  getStorageStats,
} from './src/services/storage';
import { initNetworkMode } from './src/services/network';
import type { Track, PlaybackContext } from './src/types';

function MainAppShell() {
  const { tokens } = useTheme();
  const insets = useSafeAreaInsets();
  const [currentRoute, setCurrentRoute] = useState<RootTab>('Home');
  const [tracks, setTracks] = useState<Track[]>([]);
  const [storageUsage, setStorageUsageStr] = useState('0.0 MB');
  const [isLoading, setIsLoading] = useState(true);
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);

  const {
    currentTrack,
    isPlaying,
    currentTime,
    duration,
    repeatMode,
    isShuffle,
    playbackContext,
    playTrack,
    togglePlay,
    seek,
    nextTrack,
    prevTrack,
    toggleRepeat,
    toggleShuffle,
    stopTrack,
    updatePlaybackContext,
  } = useAudioPlayer();

  const refreshLibrary = useCallback(async () => {
    try {
      const storedTracks = await getAllTracks();
      setTracks(storedTracks);
      const stats = await getStorageStats();
      setStorageUsageStr(stats.formattedSize);

      if (updatePlaybackContext) {
        updatePlaybackContext(playbackContext, storedTracks);
      }
    } catch (err) {
      console.error('[App] Failed to refresh library:', err);
    }
  }, [playbackContext, updatePlaybackContext]);

  useEffect(() => {
    const bootstrap = async () => {
      try {
        await initNetworkMode();
        await refreshLibrary();
      } finally {
        setIsLoading(false);
      }
    };

    bootstrap();
  }, [refreshLibrary]);

  const handleTrackSaved = async (newTrack: Track) => {
    try {
      await saveTrack(newTrack);
      const updatedList = await getAllTracks();
      setTracks(updatedList);
      const stats = await getStorageStats();
      setStorageUsageStr(stats.formattedSize);
    } catch (err) {
      console.error('[App] Error saving newly downloaded track:', err);
    }
  };

  const handleDeleteTrack = async (id: string) => {
    if (currentTrack?.id === id) {
      stopTrack();
    }
    await deleteTrack(id);
    await refreshLibrary();
  };

  const handleToggleFavorite = async (id: string) => {
    await toggleTrackFavorite(id);

    setTracks((prevTracks) => {
      const updated = prevTracks.map((t) =>
        t.id === id ? { ...t, isFavorite: !t.isFavorite } : t
      );

      const currentQueue =
        playbackContext === 'Favorites'
          ? updated.filter((t) => t.isFavorite)
          : updated.filter((t) => !t.isFavorite);

      if (updatePlaybackContext) {
        updatePlaybackContext(playbackContext, currentQueue);
      }
      return updated;
    });

    await refreshLibrary();
  };

  const handlePlayTrack = (
    track: Track,
    context: PlaybackContext,
    queue: Track[]
  ) => {
    playTrack(track, context, queue);
  };

  return (
    <View
      style={[
        styles.root,
        {
          backgroundColor: tokens.background,
          paddingTop: insets.top,
          paddingLeft: insets.left,
          paddingRight: insets.right,
        },
      ]}
    >
      <StatusBar
        barStyle="light-content"
        {...(Platform.OS === 'android' ? { backgroundColor: tokens.background } : {})}
      />

      <View style={styles.body}>
        {isLoading ? (
          <View style={styles.centerLoading}>
            <ActivityIndicator size="large" color={tokens.primary} />
          </View>
        ) : (
          <>
            {/* Kept permanently mounted in native view hierarchy */}
            <View
              style={[
                styles.tabContainer,
                {
                  opacity: currentRoute === 'Home' ? 1 : 0,
                  zIndex: currentRoute === 'Home' ? 1 : -1,
                  pointerEvents: currentRoute === 'Home' ? 'auto' : 'none',
                },
              ]}
            >
              <HomeScreen
                tracks={tracks}
                currentTrack={currentTrack}
                isPlaying={isPlaying}
                onPlayTrack={handlePlayTrack}
                onTogglePlay={togglePlay}
                onToggleFavorite={handleToggleFavorite}
                onDeleteTrack={handleDeleteTrack}
                onNavigateTab={setCurrentRoute}
              />
            </View>

            <View
              style={[
                styles.tabContainer,
                {
                  opacity: currentRoute === 'Download' ? 1 : 0,
                  zIndex: currentRoute === 'Download' ? 1 : -1,
                  pointerEvents: currentRoute === 'Download' ? 'auto' : 'none',
                },
              ]}
            >
              <SearchScreen onTrackSaved={handleTrackSaved} />
            </View>

            <View
              style={[
                styles.tabContainer,
                {
                  opacity: currentRoute === 'Library' ? 1 : 0,
                  zIndex: currentRoute === 'Library' ? 1 : -1,
                  pointerEvents: currentRoute === 'Library' ? 'auto' : 'none',
                },
              ]}
            >
              <LibraryScreen
                tracks={tracks}
                currentTrack={currentTrack}
                isPlaying={isPlaying}
                storageUsage={storageUsage}
                onPlayTrack={handlePlayTrack}
                onToggleFavorite={handleToggleFavorite}
                onDeleteTrack={handleDeleteTrack}
                onRefreshLibrary={refreshLibrary}
              />
            </View>
          </>
        )}
      </View>

      {/* Full Player & Mini Player Sheet */}
      <FullPlayerSheet
        currentTrack={currentTrack}
        isPlaying={isPlaying}
        currentTime={currentTime}
        duration={duration}
        repeatMode={repeatMode}
        isShuffle={isShuffle}
        playbackContext={playbackContext}
        onTogglePlay={togglePlay}
        onSeek={seek}
        onNext={nextTrack}
        onPrev={prevTrack}
        onToggleRepeat={toggleRepeat}
        onToggleShuffle={toggleShuffle}
        onToggleFavorite={handleToggleFavorite}
        onNavigateToLibrary={() => setCurrentRoute('Library')}
      />

      {/* Floating Bottom Navigation Bar */}
      <BottomNavBar
        currentRoute={currentRoute}
        onNavigate={setCurrentRoute}
        onOpenSettings={() => setIsSettingsOpen(true)}
      />

      {/* Slide-Up Settings Sheet */}
      <SettingsSheet
        visible={isSettingsOpen}
        onClose={() => setIsSettingsOpen(false)}
        storageUsage={storageUsage}
        onClearCache={refreshLibrary}
      />
    </View>
  );
}

export default function App() {
  return (
    <SafeAreaProvider>
      <ThemeProvider>
        <MainAppShell />
      </ThemeProvider>
    </SafeAreaProvider>
  );
}

} from 'react-native-reanimated';

interface MarqueeTextProps {
  text: string;
  style?: TextStyle;
  speed?: number;
}

export const MarqueeText: React.FC<MarqueeTextProps> = ({ text, style, speed = 4000 }) => {
  const translateX = useSharedValue(0);

  useEffect(() => {
    if (text.length > 25) {
      translateX.value = withRepeat(
        withTiming(-120, { duration: speed, easing: Easing.linear }),

  const [targetSeekFraction, setTargetSeekFraction] = useState<number | null>(null);
  const lastSeekFinishedTime = useRef(0);

  useEffect(() => {
    setIsFavoriteOptimistic(currentTrack?.isFavorite ?? false);
  }, [currentTrack?.id, currentTrack?.isFavorite]);

  useEffect(() => {
    setTargetSeekFraction(null);
    lastSeekFinishedTime.current = 0;
  }, [currentTrack?.id]);

  const totalDuration = duration || currentTrack?.duration || 1;
  const actualFraction = Math.min(Math.max(0, currentTime / totalDuration), 1);

  useEffect(() => {
    if (targetSeekFraction !== null) {
      const timeSinceSeek = Date.now() - lastSeekFinishedTime.current;
      const diff = Math.abs(actualFraction - targetSeekFraction);
      if (timeSinceSeek > 5000 || diff < 0.04) {
        setTargetSeekFraction(null);
      }
    }
  }, [currentTime, actualFraction, targetSeekFraction]);

  const activeProgressFraction = targetSeekFraction !== null ? targetSeekFraction : actualFraction;

  const prevWeight = useRef(new Animated.Value(1)).current;
  const playWeight = useRef(new Animated.Value(1)).current;
  const nextWeight = useRef(new Animated.Value(1)).current;

  const navBarOffset = (insets.bottom > 0 ? insets.bottom + 4 : 12) + 58 + 12;

  const expandSheet = () => {
    isExpandedRef.current = true;
    setIsExpandedState(true);
    Animated.spring(expansionFraction, {
      toValue: 1,
      tension: 70,
      friction: 9,
      useNativeDriver: false,
    }).start();
  };

  const collapseSheet = (callback?: () => void) => {
    isExpandedRef.current = false;
    Animated.timing(expansionFraction, {
      toValue: 0,
      duration: 220,
      useNativeDriver: false,
    }).start(() => {
      setIsExpandedState(false);
      if (callback) callback();
    });
  };

  const handleOpenLibrary = () => {
    collapseSheet(() => {
      if (onNavigateToLibrary) {
        onNavigateToLibrary();
      }
    });
  };

  const handleToggleFavoritePress = () => {
    if (!currentTrack) return;
    triggerIconBounce(favoriteBounceAnim);
    const nextVal = !isFavoriteOptimistic;
    setIsFavoriteOptimistic(nextVal);
    if (onToggleFavorite) {
      onToggleFavorite(currentTrack.id);
    }
  };

  const handleToggleShufflePress = () => {
    triggerIconBounce(shuffleBounceAnim);
    if (onToggleShuffle) {
      onToggleShuffle();
    }
  };

  const handleToggleRepeatPress = () => {
    triggerIconBounce(repeatBounceAnim);
    onToggleRepeat();
  };

  const handleSeekCommit = (fraction: number) => {
    setTargetSeekFraction(fraction);
    lastSeekFinishedTime.current = Date.now();
    onSeek(Math.floor(fraction * totalDuration));
  };

  const triggerButtonExpansion = (button: 'prev' | 'play' | 'next') => {
    Animated.parallel([
      Animated.spring(prevWeight, {
        toValue: button === 'prev' ? 1.15 : 0.85,
        tension: 180,
        friction: 12,
        useNativeDriver: false,
      }),
      Animated.spring(playWeight, {
        toValue: button === 'play' ? 1.15 : 0.85,
        tension: 180,
        friction: 12,
        useNativeDriver: false,
      }),
      Animated.spring(nextWeight, {
        toValue: button === 'next' ? 1.15 : 0.85,
        tension: 180,
        friction: 12,
        useNativeDriver: false,
      }),
    ]).start(() => {
      Animated.parallel([
        Animated.spring(prevWeight, { toValue: 1, tension: 140, friction: 10, useNativeDriver: false }),
        Animated.spring(playWeight, { toValue: 1, tension: 140, friction: 10, useNativeDriver: false }),
        Animated.spring(nextWeight, { toValue: 1, tension: 140, friction: 10, useNativeDriver: false }),
      ]).start();
    });
  };

  const panResponder = useRef(
    PanResponder.create({
      onMoveShouldSetPanResponder: (_, gesture) => {
        return (
          Math.abs(gesture.dy) > 15 &&
          Math.abs(gesture.dy) > Math.abs(gesture.dx) * 1.8
        );
      },
      onPanResponderMove: (_, gesture) => {
        if (!isExpandedRef.current) {
          const progress = Math.min(Math.max(-gesture.dy / (SCREEN_HEIGHT * 0.5), 0), 1);
          expansionFraction.setValue(progress);
        } else {
          const progress = Math.min(Math.max(1 - gesture.dy / (SCREEN_HEIGHT * 0.5), 0), 1);
          expansionFraction.setValue(progress);
        }
      },
      onPanResponderRelease: (_, gesture) => {
        if (!isExpandedRef.current) {
          if (gesture.dy < -70 || gesture.vy < -0.4) {
            expandSheet();
          } else {
            collapseSheet();
          }
        } else {
          if (gesture.dy > 80 || gesture.vy > 0.4) {
            collapseSheet();
          } else {
            expandSheet();
          }
        }
      },
    })
  ).current;

  if (!currentTrack) return null;

  const sheetHeight = expansionFraction.interpolate({
    inputRange: [0, 1],
    outputRange: [MINI_HEIGHT, SCREEN_HEIGHT],
  });

  const sheetBottom = expansionFraction.interpolate({
    inputRange: [0, 1],
    outputRange: [navBarOffset, 0],
  });

  const sheetBorderRadius = expansionFraction.interpolate({
    inputRange: [0, 1],
    outputRange: [shapes.sheet, 0],
  });

  const miniPlayerOpacity = expansionFraction.interpolate({
    inputRange: [0, 0.2],
    outputRange: [1, 0],
    extrapolate: 'clamp',
  });

  const fullPlayerOpacity = expansionFraction.interpolate({
    inputRange: [0.25, 1],
    outputRange: [0, 1],
    extrapolate: 'clamp',
  });

  const rootBackgroundColor = expansionFraction.interpolate({
    inputRange: [0, 0.25, 1],
    outputRange: [tokens.surface, tokens.surface, '#0D0814'],
  });

  const rawContextString = String(playbackContext ?? '').toLowerCase();
  const categoryLabel =
    rawContextString === 'favorites' || rawContextString === 'favorite'
      ? 'FAVORITES'
      : rawContextString === 'all'
      ? 'ALL TRACKS'
      : String(playbackContext ?? '').toUpperCase();

  return (
    <Animated.View
      style={[
        styles.sheetRoot,
        {
          height: sheetHeight,
          bottom: sheetBottom,
          borderRadius: sheetBorderRadius,
          backgroundColor: rootBackgroundColor,
          borderColor: isExpandedState ? 'transparent' : tokens.outline,
          borderWidth: isExpandedState ? 0 : 1.5,
          left: isExpandedState ? 0 : 16,
          right: isExpandedState ? 0 : 16,
        },
      ]}
      {...panResponder.panHandlers}
    >
      <Animated.View
        style={[
          styles.blurredBackdropContainer,
          { opacity: fullPlayerOpacity },
        ]}
        pointerEvents="none"
      >
        {currentTrack.thumbnailUrl ? (
          <Image
            source={{ uri: currentTrack.thumbnailUrl }}
            style={styles.blurredBackdropImage}
            blurRadius={Platform.OS === 'ios' ? 18 : 10}
          />
        ) : null}

        <View style={[styles.glassFrostedScrim, { backgroundColor: '#130B2288' }]} />
      </Animated.View>

      {/* Mini Player */}
      <Animated.View
        style={[styles.miniLayer, { opacity: miniPlayerOpacity }]}
        pointerEvents={isExpandedState ? 'none' : 'auto'}
      >
        <TouchableOpacity activeOpacity={1} onPress={expandSheet} style={styles.miniTouchable}>
          <View style={styles.miniProgressTrack}>
            <View
              style={[
                styles.miniProgressFill,
                { width: `${activeProgressFraction * 100}%`, backgroundColor: tokens.primary },
              ]}
            />
          </View>

          <View style={styles.miniContentRow}>
            <View
              style={[
                styles.miniArtContainer,
                { backgroundColor: tokens.surfaceVariant, borderRadius: shapes.md },
              ]}
            >
              {currentTrack.thumbnailUrl ? (
                <Image source={{ uri: currentTrack.thumbnailUrl }} style={styles.fullImage} />
              ) : (
                <Icon name="library" size={18} color={tokens.primary} />
              )}
            </View>

            <View style={styles.miniMetaCol}>
              <Text numberOfLines={1} style={[styles.miniTitle, { color: tokens.onSurface }]}>
                {currentTrack.title}
              </Text>
              <Text numberOfLines={1} style={[styles.miniSubtitle, { color: tokens.onSurfaceVariant }]}>
                {currentTrack.artist}
              </Text>
            </View>

            <M3Pressable
              onPress={onTogglePlay}
              style={[styles.miniPlayCircle, { backgroundColor: tokens.primary }]}
            >
              <Icon name={isPlaying ? 'pause' : 'play'} size={20} color="#FFFFFF" />

          </View>
        </TouchableOpacity>

      {/* Full Player */}
      <Animated.View

        {/* Top App Bar Header */}
        <View style={styles.topAppBarRow}>
          <View style={styles.topLeftHeaderGroup}>

              onPress={() => collapseSheet()}

            >
              <Icon name="chevron-down" size={24} color={tokens.primary} />
            </M3Pressable>

            <View style={styles.nowPlayingTextGroup}>
              <Text style={[styles.nowPlayingText, { color: tokens.onSurface }]}>
                Now Playing
              </Text>
              <Text style={[styles.categoryBadgeText, { color: tokens.primary }]}>
                {categoryLabel}
              </Text>
            </View>
          </View>

          <M3Pressable
            onPress={handleOpenLibrary}
            style={[styles.glassCircleButton, { borderColor: 'rgba(255, 255, 255, 0.14)' }]}
          >
            <Icon name="queue" size={20} color={tokens.primary} />
          </M3Pressable>

        {/* Hero Artwork Section */}
        <View style={styles.artworkSection}>
          <View

        <View style={styles.scrubberSection}>
          <WavySliderExpressive
            progressFraction={activeProgressFraction}
            isPlaying={isPlaying}
            onSeekChange={(fraction) => setTargetSeekFraction(fraction)}
            onSeekCommit={handleSeekCommit}
            activeTrackColor={tokens.primary}

          <View style={styles.timeLabelRow}>
            <Text style={[styles.timeLabelText, { color: tokens.onSurfaceVariant }]}>

            onPress={handleToggleFavoritePress}

    </Animated.View>
  );
};

import React, { useState, useRef, useMemo } from 'react';
import {
  View,
  Text,
  TextInput,
  ScrollView,
  Image,
  StyleSheet,
  Dimensions,
  Animated,
  Easing,
} from 'react-native';
import { useTheme } from '../../theme/ThemeContext';
import { Icon } from '../../components/common/Icon';
import { SongCard } from '../../components/cards/SongCard';
import { M3Pressable } from '../../components/common/M3Pressable';
import type { Track, PlaybackContext } from '../../types';

const { width: SCREEN_WIDTH } = Dimensions.get('window');
const CAROUSEL_CARD_WIDTH = Math.min(SCREEN_WIDTH * 0.52, 200);

interface HomeScreenProps {
  tracks: Track[];
  currentTrack: Track | null;
  isPlaying: boolean;
  onPlayTrack: (track: Track, context: PlaybackContext, queue: Track[]) => void;
  onTogglePlay: () => void;
  onToggleFavorite: (id: string) => void;
  onDeleteTrack: (id: string) => void;
  onNavigateTab: (tab: 'Download' | 'Library') => void;
}

export const HomeScreen: React.FC<HomeScreenProps> = ({
  tracks,
  currentTrack,
  isPlaying,
  onPlayTrack,
  onTogglePlay,
  onToggleFavorite,
  onDeleteTrack,
  onNavigateTab,
}) => {
  const { tokens, shapes } = useTheme();

  const [isSearchOpen, setIsSearchOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const searchInputRef = useRef<TextInput>(null);

  const morphProgress = useRef(new Animated.Value(0)).current;

  const favoriteTracks = tracks.filter((t) => t.isFavorite);
  const recentTracks = tracks.slice(0, 6);
  const featuredTracks = tracks.slice(0, 6);

  const searchResults = useMemo(() => {
    if (!searchQuery.trim()) return [];
    const q = searchQuery.toLowerCase().trim();
    return tracks.filter(
      (t) =>
        t.title.toLowerCase().includes(q) ||
        t.artist.toLowerCase().includes(q)
    );
  }, [tracks, searchQuery]);

  const openSearch = () => {
    setIsSearchOpen(true);
    Animated.timing(morphProgress, {
      toValue: 1,
      duration: 280,
      easing: Easing.bezier(0.2, 0.0, 0.0, 1.0),
      useNativeDriver: false,
    }).start(() => {
      searchInputRef.current?.focus();
    });
  };

  const closeSearch = () => {
    searchInputRef.current?.blur();
    Animated.timing(morphProgress, {
      toValue: 0,
      duration: 220,
      easing: Easing.bezier(0.4, 0.0, 1.0, 1.0),
      useNativeDriver: false,
    }).start(() => {

  };

    inputRange: [0.15, 1],
    outputRange: [0, 1],

    inputRange: [0, 1],
    outputRange: [0.94, 1],

          {searchResults.length === 0 ? (

              <Icon name="library" size={32} color={tokens.outline} />

                key={track.id}
                song={track}

                onPress={() => {

                onDelete={() => onDeleteTrack(track.id)}

          {/* ─── 3. Quick Action Chips with M3 Press ─── */}

                  backgroundColor: tokens.surface,

                      </M3Pressable>

                      <Text numberOfLines={1} style={[styles.carouselTitle, { color: tokens.onSurface }]}>
                        {item.title}

                        {item.artist}
                      </Text>
                    </View>
                  </M3Pressable>
                );
              })}
            </ScrollView>
          )}

          {/* ─── 5. Recently Added Tracks ─── */}
          <View style={[styles.sectionHeader, { marginTop: 22 }]}>
            <Text style={[styles.sectionTitle, { color: tokens.onSurface }]}>
              Recently Added
            </Text>
            <M3Pressable onPress={() => onNavigateTab('Library')}>
              <Text style={[styles.seeAllLink, { color: tokens.primary }]}>View Library</Text>
            </M3Pressable>
          </View>

          <View style={styles.recentList}>
            {recentTracks.map((track) => (
              <SongCard
                key={track.id}
                song={track}
                isPlaying={currentTrack?.id === track.id && isPlaying}
                onPress={() => handleTrackCardPress(track)}
                onToggleFavorite={() => onToggleFavorite(track.id)}
                onDelete={() => onDeleteTrack(track.id)}
              />
            ))}
          </View>
        </>
      )}
    </ScrollView>
  );
};

import React, { useState, useMemo } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  ScrollView,
  StyleSheet,
  ActivityIndicator,
} from 'react-native';
import { useTheme } from '../../theme/ThemeContext';
import { Icon } from '../../components/common/Icon';
import { SongCard } from '../../components/cards/SongCard';
import { pruneStaleStorage } from '../../services/storage';
import type { Track, PlaybackContext } from '../../types';

interface LibraryScreenProps {
  tracks: Track[];
  currentTrack: Track | null;
  isPlaying: boolean;
  storageUsage: string;
  onPlayTrack: (track: Track, context: PlaybackContext, queue: Track[]) => void;
  onToggleFavorite: (id: string) => void;
  onDeleteTrack: (id: string) => void;
  onRefreshLibrary: () => Promise<void>;
}

type LibraryTabFilter = 'ALL' | 'FAVORITES' | 'STORAGE';

export const LibraryScreen: React.FC<LibraryScreenProps> = ({
  tracks,
  currentTrack,
  isPlaying,
  storageUsage,
  onPlayTrack,
  onToggleFavorite,
  onDeleteTrack,
  onRefreshLibrary,
}) => {
  const { tokens, shapes } = useTheme();

  const [activeTab, setActiveTab] = useState<LibraryTabFilter>('ALL');
  const [searchQuery, setSearchQuery] = useState('');
  const [isCleaning, setIsCleaning] = useState(false);
  const [cleanNotice, setCleanNotice] = useState<string | null>(null);

  // Exclusive separation: Favorites move out of the All Tracks list
  const nonFavoriteTracks = useMemo(
    () => tracks.filter((t) => !t.isFavorite),
    [tracks]
  );
  const favoriteTracks = useMemo(
    () => tracks.filter((t) => t.isFavorite),
    [tracks]
  );

  // Tab Selection
  const filteredByTab = useMemo(() => {
    if (activeTab === 'FAVORITES') {
      return favoriteTracks;
    }
    return nonFavoriteTracks;
  }, [nonFavoriteTracks, favoriteTracks, activeTab]);

  // Search Query Filtering
  const displayedTracks = useMemo(() => {
    if (!searchQuery.trim()) return filteredByTab;
    const q = searchQuery.toLowerCase();
    return filteredByTab.filter(
      (t) =>
        t.title.toLowerCase().includes(q) ||
        t.artist.toLowerCase().includes(q)
    );
  }, [filteredByTab, searchQuery]);

  const handlePruneStorage = async () => {
    if (isCleaning) return;
    setIsCleaning(true);
    setCleanNotice(null);
    try {
      const res = await pruneStaleStorage();
      await onRefreshLibrary();
      setCleanNotice(
        res.prunedCount > 0
          ? `Cleaned ${res.prunedCount} orphaned file(s) successfully.`
          : 'Storage is clean. No orphaned files detected.'
      );
    } catch {
      setCleanNotice('Failed to complete storage cleanup.');
    } finally {
      setIsCleaning(false);
      setTimeout(() => setCleanNotice(null), 4000);
    }
  };

  return (
    <ScrollView
      contentContainerStyle={styles.container}
      showsVerticalScrollIndicator={false}
      keyboardShouldPersistTaps="handled"
    >
      {/* Screen Header */}
      <View style={styles.headerRow}>
        <View>
          <Text style={[styles.title, { color: tokens.onSurface }]}>Your Library</Text>
          <Text style={[styles.subtitle, { color: tokens.onSurfaceVariant }]}>
            {tracks.length} offline {tracks.length === 1 ? 'song' : 'songs'} saved
          </Text>
        </View>

        {/* Storage Size Badge */}
        <View
          style={[
            styles.storageBadge,
            {
              backgroundColor: tokens.surfaceVariant,
              borderRadius: shapes.pill,
              borderColor: tokens.outline,
            },
          ]}
        >
          <Icon name="library" size={14} color={tokens.primary} />
          <Text style={[styles.storageText, { color: tokens.onSurface }]}>{storageUsage}</Text>
        </View>
      </View>

      {/* Search Input Bar */}
      <View style={styles.searchWrapper}>
        <View style={styles.searchIcon}>
          <Icon name="search" size={18} color={tokens.outline} />
        </View>
        <TextInput
          value={searchQuery}
          onChangeText={setSearchQuery}
          placeholder="Filter songs or artists..."
          placeholderTextColor={tokens.onSurfaceVariant}
          autoCapitalize="none"
          autoCorrect={false}
          style={[
            styles.searchInput,
            {
              backgroundColor: tokens.surfaceVariant,
              color: tokens.onSurface,
              borderRadius: shapes.md,
            },
          ]}
        />
      </View>

      {/* Sub-Tabs Row */}
      <View style={styles.pillRow}>
        <TouchableOpacity
          onPress={() => setActiveTab('ALL')}
          style={[
            styles.pill,
            {
              borderRadius: shapes.pill,
              backgroundColor: activeTab === 'ALL' ? tokens.primary : tokens.surfaceVariant,
            },
          ]}
        >
          <Text
            style={[
              styles.pillText,
              { color: activeTab === 'ALL' ? '#FFFFFF' : tokens.onSurfaceVariant },
            ]}
          >
            All Tracks ({nonFavoriteTracks.length})
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          onPress={() => setActiveTab('FAVORITES')}
          style={[
            styles.pill,
            {
              borderRadius: shapes.pill,
              backgroundColor:
                activeTab === 'FAVORITES' ? tokens.primary : tokens.surfaceVariant,
            },
          ]}
        >
          <Text
            style={[
              styles.pillText,
              { color: activeTab === 'FAVORITES' ? '#FFFFFF' : tokens.onSurfaceVariant },
            ]}
          >
            Favorites ({favoriteTracks.length})
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          onPress={() => setActiveTab('STORAGE')}
          style={[
            styles.pill,
            {
              borderRadius: shapes.pill,
              backgroundColor:
                activeTab === 'STORAGE' ? tokens.primary : tokens.surfaceVariant,
            },
          ]}
        >
          <Text
            style={[
              styles.pillText,
              { color: activeTab === 'STORAGE' ? '#FFFFFF' : tokens.onSurfaceVariant },
            ]}
          >
            Storage
          </Text>
        </TouchableOpacity>
      </View>

      {/* Tab 3 View: Storage Manager */}
      {activeTab === 'STORAGE' ? (
        <View
          style={[
            styles.storageCard,
            {
              backgroundColor: tokens.surfaceVariant,
              borderRadius: shapes.lg,
            },
          ]}
        >
          <Text style={[styles.cardTitle, { color: tokens.onSurface }]}>Disk Management</Text>
          <Text style={[styles.cardDesc, { color: tokens.onSurfaceVariant }]}>
            All audio files are saved in the permanent app sandbox. You can clean up broken or
            incomplete downloads to free up storage space.
          </Text>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { color: tokens.onSurfaceVariant }]}>Total Audio Space</Text>
            <Text style={[styles.statValue, { color: tokens.primary }]}>{storageUsage}</Text>
          </View>

          <View style={styles.statRow}>
            <Text style={[styles.statLabel, { color: tokens.onSurfaceVariant }]}>Downloaded Tracks</Text>
            <Text style={[styles.statValue, { color: tokens.onSurface }]}>{tracks.length}</Text>
          </View>

          {cleanNotice && (
            <View style={styles.noticeBox}>
              <Text style={styles.noticeText}>{cleanNotice}</Text>
            </View>
          )}

          <TouchableOpacity
            onPress={handlePruneStorage}
            disabled={isCleaning}
            style={[
              styles.cleanupBtn,
              {
                backgroundColor: tokens.primary,
                borderRadius: shapes.pill,
                opacity: isCleaning ? 0.6 : 1,
              },
            ]}
          >
            {isCleaning ? (
              <ActivityIndicator size="small" color="#FFFFFF" />
            ) : (
              <Text style={styles.cleanupBtnText}>Prune Orphaned Files</Text>
            )}
          </TouchableOpacity>
        </View>
      ) : (
        /* Tabs 1 & 2 View: Track List */
        <View style={styles.trackListContainer}>
          {displayedTracks.length === 0 ? (
            <View style={styles.emptyContainer}>
              <Icon name="library" size={48} color={tokens.outline} />
              <Text style={[styles.emptyTitle, { color: tokens.onSurface }]}>
                {activeTab === 'FAVORITES' ? 'No favorites yet' : 'No un-favorited tracks'}
              </Text>
              <Text style={[styles.emptySubtitle, { color: tokens.onSurfaceVariant }]}>
                {activeTab === 'FAVORITES'
                  ? 'Tap the heart icon on any track to add it to your favorites.'
                  : 'All your tracks are currently marked as favorites.'}
              </Text>
            </View>
          ) : (
            displayedTracks.map((track) => (
              <SongCard
                key={track.id}
                song={track}
                isPlaying={currentTrack?.id === track.id && isPlaying}
                onPress={() =>
                  onPlayTrack(
                    track,
                    activeTab === 'FAVORITES' ? 'Favorite' : 'All',
                    displayedTracks
                  )
                }
                onToggleFavorite={() => onToggleFavorite(track.id)}
                onDelete={() => onDeleteTrack(track.id)}
              />
            ))
          )}
        </View>
      )}
    </ScrollView>
  );
};

import * as Haptics from 'expo-haptics';

interface SearchScreenProps {
  onTrackSaved: (track: Track) => Promise<void>;

interface QueueSongItemProps {
  item: QueueItem;
  tokens: any;
  shapes: any;
  onCancel: (id: string) => void;
  onResume: (id: string) => void;
  onRemove: (id: string) => void;
  onCompleteFadeOut: (id: string) => void;
}

// Strict YouTube URL validator helper
export const isValidYouTubeUrl = (testUrl: string): boolean => {
  if (!testUrl || typeof testUrl !== 'string') return false;
  const trimmed = testUrl.trim();
  const ytRegex =
    /^(https?:\/\/)?((www|m|music)\.)?(youtube\.com\/(watch\?.*v=|shorts\/|embed\/)|youtu\.be\/)[\w-]{11}/i;
  return ytRegex.test(trimmed);
};

const QueueSongItem: React.FC<QueueSongItemProps> = ({
  item,
  tokens,
  shapes,
  onCancel,
  onResume,
  onRemove,
  onCompleteFadeOut,
}) => {
  const fadeAnim = useRef(new Animated.Value(1)).current;
  const isDownloading = item.status === 'downloading';
  const isResolving = item.status === 'fetching_info';

  useEffect(() => {
    if (item.status === 'completed') {
      const timer = setTimeout(() => {
        Animated.timing(fadeAnim, {
          toValue: 0,
          duration: 900,
          useNativeDriver: true,
        }).start(() => {
          onCompleteFadeOut(item.id);
        });
      }, 700);

      return () => clearTimeout(timer);
    }
  }, [item.status, item.id, fadeAnim, onCompleteFadeOut]);

  return (
    <Animated.View
      style={[
        styles.songCardContainer,
        {
          opacity: fadeAnim,
          transform: [
            {
              scale: fadeAnim.interpolate({
                inputRange: [0, 1],
                outputRange: [0.96, 1],
              }),
            },
          ],
          backgroundColor: isDownloading ? tokens.surfaceVariant : tokens.surface,
          borderRadius: shapes.md,
          borderColor: isDownloading ? tokens.primary : 'rgba(255, 255, 255, 0.05)',
        },
      ]}
    >
      <View
        style={[
          styles.artWrapper,
          {
            borderRadius: shapes.sm,
            backgroundColor: tokens.surfaceVariant,
          },
        ]}
      >
        {item.thumbnail ? (
          <Image source={{ uri: item.thumbnail }} style={styles.artImage} />
        ) : isResolving ? (
          <ActivityIndicator size="small" color={tokens.primary} />
        ) : (
          <View style={styles.artFallback}>
            <Icon name="library" size={20} color={tokens.primary} />
          </View>
        )}
      </View>

      <View style={styles.metaContainer}>
        <Text
          numberOfLines={1}
          style={[styles.queueTrackTitle, { color: isDownloading ? tokens.primary : tokens.onSurface }]}
        >
          {item.title || item.url}
        </Text>

        <Text numberOfLines={1} style={[styles.artistSubtitle, { color: tokens.onSurfaceVariant }]}>
          {isResolving ? (
            'Fetching song details...'
          ) : isDownloading ? (
            `Downloading • ${item.progressPercent}%`
          ) : item.status === 'completed' ? (
            `Downloaded • ${item.durationFormatted && item.durationFormatted !== '--:--' ? item.durationFormatted : 'Saved'}`
          ) : item.status === 'paused' ? (
            'Paused • Tap to resume'
          ) : item.status === 'error' ? (
            item.statusText
          ) : item.channel ? (
            `${item.channel} • In Queue`
          ) : (
            'Queued'
          )}
        </Text>
      </View>

      <View style={styles.actionRow}>
        {isDownloading ? (
          <TouchableOpacity onPress={() => onCancel(item.id)} style={styles.iconBtn}>
            <Icon name="pause" size={17} color="#F87171" />
          </TouchableOpacity>
        ) : item.status === 'paused' || item.status === 'error' ? (
          <TouchableOpacity onPress={() => onResume(item.id)} style={styles.iconBtn}>
            <Icon name="play" size={17} color={tokens.primary} />
          </TouchableOpacity>
        ) : null}

        <TouchableOpacity onPress={() => onRemove(item.id)} style={styles.iconBtn}>
          <Icon name="close" size={16} color={tokens.onSurfaceVariant} />
        </TouchableOpacity>
      </View>

      {isDownloading ? (
        <View style={styles.flushProgressTrack}>
          <View
            style={[
              styles.flushProgressFill,
              {
                width: `${item.progressPercent}%`,
                backgroundColor: tokens.primary,
              },
            ]}
          />
        </View>
      ) : null}
    </Animated.View>
  );
};

export const SearchScreen: React.FC<SearchScreenProps> = ({ onTrackSaved }) => {
  const { tokens, shapes } = useTheme();

  const [url, setUrl] = useState('');
  const [loading, setLoading] = useState(false);
  const [preview, setPreview] = useState<VideoInfo | null>(null);
  const [error, setError] = useState<string | null>(null);

  // Duplicate checks
  const [existingDuplicate, setExistingDuplicate] = useState<Track | null>(null);
  const [isDuplicateInQueue, setIsDuplicateInQueue] = useState(false);

  const [queue, setQueue] = useState<QueueItem[]>([]);
  const [selectedIds, setSelectedIds] = useState<string[]>([]);

  useEffect(() => {
    queueService.registerTrackSavedCallback(onTrackSaved);
    const unsubscribe = queueService.subscribe((updatedQueue) => {
      setQueue(updatedQueue);
    });
    return () => unsubscribe();
  }, [onTrackSaved]);

  const triggerHaptic = (style: Haptics.ImpactFeedbackStyle = Haptics.ImpactFeedbackStyle.Light) => {
    try {
      Haptics.impactAsync(style);
    } catch {}
  };

  const handlePasteFromClipboard = async () => {
    try {
      triggerHaptic(Haptics.ImpactFeedbackStyle.Medium);
      const text = await Clipboard.getStringAsync();
      const clean = text ? text.trim() : '';

      if (!clean) {
        setError('Clipboard is empty.');
        return;
      }

      setUrl(clean);
      setPreview(null);
      setExistingDuplicate(null);
      setIsDuplicateInQueue(false);

      if (!isValidYouTubeUrl(clean)) {
        setError('Pasted content is not a valid YouTube URL.');
      } else {
        setError(null);
      }
    } catch {
      setError('Could not access clipboard.');
    }
  };

  const handleFetchInfo = async () => {
    const cleanUrl = url.trim();
    if (!cleanUrl) return;

    if (!isValidYouTubeUrl(cleanUrl)) {
      triggerHaptic(Haptics.ImpactFeedbackStyle.Medium);
      setError('Invalid YouTube link. Please paste a valid youtube.com or youtu.be URL.');
      return;
    }

    setLoading(true);
    setError(null);
    setPreview(null);
    setExistingDuplicate(null);
    setIsDuplicateInQueue(false);

    try {
      const info = await fetchVideoInfo(cleanUrl);
      setPreview(info);

      const duplicateInDb = await findDuplicateTrack({
        id: info.id,
        title: info.title,
        channel: info.channel,
        thumbnail: info.thumbnail,
      });

      const currentQueue = queueService.getQueue();
      const inQueue = currentQueue.some(
        (q) =>
          q.url === info.originalUrl ||
          (info.id && q.id?.includes(info.id)) ||
          (q.title && info.title && q.title.trim().toLowerCase() === info.title.trim().toLowerCase())
      );

      if (duplicateInDb) {
        setExistingDuplicate(duplicateInDb);
        triggerHaptic(Haptics.ImpactFeedbackStyle.Medium);
      } else if (inQueue) {
        setIsDuplicateInQueue(true);
        triggerHaptic(Haptics.ImpactFeedbackStyle.Medium);
      }
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Failed to fetch video information';
      setError(msg);
    } finally {
      setLoading(false);
    }
  };

  const handleAddToQueue = (videoData?: VideoInfo) => {
    const targetUrl = videoData?.originalUrl || url.trim();
    if (!targetUrl) return;

    if (!isValidYouTubeUrl(targetUrl)) {
      setError('Cannot queue: invalid YouTube URL.');
      return;
    }

    triggerHaptic(Haptics.ImpactFeedbackStyle.Medium);
    queueService.addToQueue(targetUrl, videoData || null);

    setUrl('');
    setPreview(null);
    setExistingDuplicate(null);
    setIsDuplicateInQueue(false);
  };

  const handleItemCompletedFadeOut = (id: string) => {
    queueService.removeItem(id);
  };

  const handleCancelItem = (id: string) => {
    triggerHaptic();
    queueService.cancelItem(id);
  };

  const handleResumeItem = (id: string) => {
    triggerHaptic();
    queueService.resumeItem(id);
  };

  const handleRemoveItem = (id: string) => {
    triggerHaptic();
    queueService.removeItem(id);
    setSelectedIds((prev) => prev.filter((selId) => selId !== id));
  };

  const handleBulkResume = () => {
    triggerHaptic();
    queueService.bulkResume(selectedIds);
    setSelectedIds([]);
  };

  const handleBulkRemove = () => {
    triggerHaptic(Haptics.ImpactFeedbackStyle.Medium);
    queueService.bulkRemove(selectedIds);
    setSelectedIds([]);
  };

  return (
    <ScrollView
      contentContainerStyle={styles.container}
      showsVerticalScrollIndicator={false}
      keyboardShouldPersistTaps="handled"
    >
      {/* ─── 1. Header (OFFLINE ENGINE / Download Queue) ─── */}
      <View style={styles.headerBox}>
        <View style={styles.headerBadge}>
          <Icon name="sparkles" size={14} color={tokens.primary} />
          <Text style={[styles.headerBadgeText, { color: tokens.primary }]}>
            OFFLINE ENGINE
          </Text>
        </View>
        <Text style={[styles.headerTitle, { color: tokens.onSurface }]}>
          Download Queue
        </Text>
        <Text style={[styles.headerSubtitle, { color: tokens.onSurfaceVariant }]}>
          Paste links to queue multiple downloads. Tracks download sequentially in background.
        </Text>
      </View>

      {/* ─── 2. Input Card Panel with Enhanced Paste Button ─── */}
      <View
        style={[
          styles.cardPanel,
          {
            backgroundColor: tokens.surface,
            borderRadius: shapes.sheet,
            borderColor: error ? '#F87171' : tokens.outline + '40',
          },
        ]}
      >
        <View style={styles.inputRow}>
          <View style={styles.searchIconBox}>
            <Icon name="search" size={20} color={tokens.onSurfaceVariant} />
          </View>

          <TextInput
            value={url}
            onChangeText={(text) => {
              setUrl(text);
              if (error) setError(null);
              setExistingDuplicate(null);
              setIsDuplicateInQueue(false);
            }}
            placeholder="Paste YouTube audio link..."
            placeholderTextColor={tokens.onSurfaceVariant + '75'}
            autoCapitalize="none"
            autoCorrect={false}
            keyboardType="url"
            returnKeyType="go"
            onSubmitEditing={handleFetchInfo}
            style={[
              styles.textInput,
              {
                backgroundColor: tokens.surfaceVariant + '35',
                color: tokens.onSurface,
                borderRadius: shapes.pill,
                borderColor: error ? '#F87171' : tokens.outline + '20',
              },
            ]}
          />

          {/* Right Action: Cross button if input is filled, otherwise Expressive Paste Chip */}
          {url.length > 0 ? (
            <TouchableOpacity
              onPress={() => {
                setUrl('');
                setPreview(null);
                setError(null);
                setExistingDuplicate(null);
                setIsDuplicateInQueue(false);
              }}
              style={styles.clearBtnAction}
            >
              <Icon name="close" size={16} color={tokens.onSurfaceVariant} />
            </TouchableOpacity>
          ) : (
            <M3Pressable
              onPress={handlePasteFromClipboard}
              pressedScale={0.92}
              style={[
                styles.expressivePasteBtn,
                {
                  backgroundColor: tokens.primaryContainer,
                  borderColor: tokens.primary + '38',
                  borderRadius: shapes.pill,
                },
              ]}
            >
              <View style={styles.pasteBtnInner}>
                <Icon name="bookmark" size={13} color={tokens.primary} />
                <Text style={[styles.expressivePasteText, { color: tokens.primary }]}>
                  PASTE
                </Text>
              </View>
            </M3Pressable>
          )}
        </View>

        {!preview && (
          <TouchableOpacity
            onPress={handleFetchInfo}
            disabled={loading || !url.trim()}
            activeOpacity={0.85}
            style={[
              styles.actionPillBtn,
              {
                backgroundColor: tokens.primary,
                borderRadius: shapes.pill,
                opacity: loading || !url.trim() ? 0.45 : 1,
              },
            ]}
          >
            {loading ? (
              <View style={styles.btnRow}>

            ) : (
              <View style={styles.btnRow}>
                <Icon name="search" size={18} color="#FFFFFF" />
                <Text style={styles.btnText}>Preview Track</Text>
              </View>
            )}
          </TouchableOpacity>
        )}

        {error && (
          <View
            style={[
              styles.errorCard,

                backgroundColor: 'rgba(239, 68, 68, 0.15)',
                borderRadius: shapes.sm,
              },

            <Text style={styles.errorText}>{error}</Text>
          </View>
        )}
      </View>

      {/* ─── 3. Preview Card Panel (With Duplicate Banner) ─── */}
      {preview && (
        <View
          style={[
            styles.cardPanel,
            {
              backgroundColor: tokens.surface,
              borderRadius: shapes.sheet,
              borderColor: existingDuplicate
                ? '#F59E0B'
                : isDuplicateInQueue
                ? tokens.primary
                : tokens.primary,

              <Icon name="shield" size={16} color="#F59E0B" />
              <Text style={styles.duplicateNoticeText}>
                This song already exists in your offline library!
              </Text>
            </View>
          )}

          {isDuplicateInQueue && !existingDuplicate && (
            <View
              style={[
                styles.duplicateNoticeBox,
                { backgroundColor: 'rgba(124, 213, 255, 0.15)', borderColor: 'rgba(124, 213, 255, 0.3)' },
              ]}
            >
              <Icon name="sparkles" size={16} color={tokens.primary} />
              <Text style={[styles.duplicateNoticeText, { color: tokens.primary }]}>
                This song is already waiting in your active queue!
              </Text>

          )}

          <View style={styles.previewMetaRow}>
            <Image
              source={{ uri: preview.thumbnail }}

import AsyncStorage from '@react-native-async-storage/async-storage';
import { fetchVideoInfo, downloadAudioStream } from './api';
import type { VideoInfo, Track } from '../types';

const QUEUE_STORAGE_KEY = 'tunes_download_queue';

export type QueueItemStatus =
  | 'fetching_info'
  | 'pending'
  | 'downloading'
  | 'paused'
  | 'completed'
  | 'error';

export interface QueueItem {
  id: string;
  url: string;
  title?: string;
  channel?: string;
  thumbnail?: string;
  durationFormatted?: string;
  status: QueueItemStatus;
  progressPercent: number;
  statusText: string;
  videoInfo?: VideoInfo | null;
  errorMessage?: string;
  createdAt: number;
}

type QueueListener = (items: QueueItem[]) => void;
type TrackSavedCallback = (track: Track) => Promise<void>;

class QueueService {
  private queue: QueueItem[] = [];
  private listeners: Set<QueueListener> = new Set();
  private isProcessing = false;
  private abortControllers: Map<string, AbortController> = new Map();
  private onTrackSavedCallback: TrackSavedCallback | null = null;
  private isInitialized = false;

  constructor() {
    this.init();
  }

  public registerTrackSavedCallback(cb: TrackSavedCallback) {
    this.onTrackSavedCallback = cb;
  }

  private async init() {
    if (this.isInitialized) return;
    this.isInitialized = true;

    try {
      const raw = await AsyncStorage.getItem(QUEUE_STORAGE_KEY);
      if (raw) {
        const parsed: QueueItem[] = JSON.parse(raw);
        // If app was killed while downloading, mark paused
        this.queue = parsed.map((item) => {
          if (item.status === 'downloading' || item.status === 'fetching_info') {
            return {
              ...item,
              status: 'paused',
              statusText: 'Paused (App closed)',
            };
          }
          return item;
        });
        this.notify();
      }
    } catch (err) {
      console.warn('[QueueService] Boot load error:', err);
    }
  }

  public getQueue(): QueueItem[] {
    return [...this.queue];
  }

  public subscribe(listener: QueueListener): () => void {
    this.listeners.add(listener);
    listener([...this.queue]);
    return () => {
      this.listeners.delete(listener);
    };
  }

  private notify() {
    const copy = [...this.queue];
    this.listeners.forEach((l) => l(copy));
    AsyncStorage.setItem(QUEUE_STORAGE_KEY, JSON.stringify(copy)).catch(() => {});
  }

  private updateItem(id: string, updates: Partial<QueueItem>) {
    this.queue = this.queue.map((item) => (item.id === id ? { ...item, ...updates } : item));
    this.notify();
  }

  public addToQueue(url: string, videoInfo?: VideoInfo | null) {
    const cleanUrl = url.trim();
    if (!cleanUrl) return;

    const hasInfoAlready = Boolean(videoInfo);
    const newId = `queue_${Date.now()}_${Math.random().toString(36).substring(2, 6)}`;

    const newItem: QueueItem = {
      id: newId,
      url: cleanUrl,
      title: videoInfo?.title || cleanUrl,
      channel: videoInfo?.channel,
      thumbnail: videoInfo?.thumbnail,
      durationFormatted: videoInfo?.durationFormatted,
      status: hasInfoAlready ? 'pending' : 'fetching_info',
      progressPercent: 5,
      statusText: hasInfoAlready ? 'In queue...' : 'Resolving info...',
      videoInfo: videoInfo || null,
      createdAt: Date.now(),
    };

    this.queue.push(newItem);
    this.notify();

    if (!hasInfoAlready) {
      this.fetchInfoImmediately(newId, cleanUrl);
    } else {
      setTimeout(() => this.processNext(), 50);
    }
  }

  private async fetchInfoImmediately(itemId: string, itemUrl: string) {
    try {
      const info = await fetchVideoInfo(itemUrl);
      this.updateItem(itemId, {
        videoInfo: info,
        title: info.title,
        channel: info.channel,
        thumbnail: info.thumbnail,
        status: 'pending',
        statusText: 'In queue...',
      });
      setTimeout(() => this.processNext(), 50);
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Failed to fetch details';
      this.updateItem(itemId, {
        status: 'error',
        errorMessage: msg,
        statusText: 'Meta fetch error',
      });
    }
  }

  public async processNext() {
    if (this.isProcessing) return;

    const nextItem = this.queue.find((item) => item.status === 'pending');
    if (!nextItem) return;

    this.isProcessing = true;
    const controller = new AbortController();
    this.abortControllers.set(nextItem.id, controller);

    try {
      this.updateItem(nextItem.id, {
        status: 'downloading',
        statusText: 'Downloading stream...',
        progressPercent: 10,
      });

      const track = await downloadAudioStream(
        nextItem.url,
        nextItem.videoInfo,
        (progress) => {
          this.updateItem(nextItem.id, {
            progressPercent: Math.max(10, progress.percent),
            statusText: progress.statusText,
          });
        },
        3,
        2000,
        controller.signal
      );

      if (this.onTrackSavedCallback) {
        await this.onTrackSavedCallback(track);
      }

      this.updateItem(nextItem.id, {
        status: 'completed',
        durationFormatted: track.durationFormatted,
        statusText: 'Saved to library',
        progressPercent: 100,
      });
    } catch (err: unknown) {
      if (controller.signal.aborted) {
        this.updateItem(nextItem.id, {
          status: 'paused',
          statusText: 'Paused by user',
        });
      } else {
        const msg = err instanceof Error ? err.message : 'Download failed';
        this.updateItem(nextItem.id, {
          status: 'error',
          errorMessage: msg,
          statusText: msg,
        });
      }
    } finally {
      this.abortControllers.delete(nextItem.id);
      this.isProcessing = false;
      setTimeout(() => this.processNext(), 150);
    }
  }

  public cancelItem(id: string) {
    const controller = this.abortControllers.get(id);
    if (controller) {
      controller.abort();
      this.abortControllers.delete(id);
    }
    this.updateItem(id, {
      status: 'paused',
      statusText: 'Paused',
    });
  }

  public resumeItem(id: string) {
    const item = this.queue.find((q) => q.id === id);
    if (!item) return;

    if (!item.videoInfo) {
      this.updateItem(id, {
        status: 'fetching_info',
        statusText: 'Retrying details...',
        errorMessage: undefined,
      });
      this.fetchInfoImmediately(id, item.url);
    } else {
      this.updateItem(id, {
        status: 'pending',
        statusText: 'Queued',
        errorMessage: undefined,
      });
      setTimeout(() => this.processNext(), 50);
    }
  }

  public removeItem(id: string) {
    const controller = this.abortControllers.get(id);
    if (controller) {
      controller.abort();
      this.abortControllers.delete(id);
    }
    this.queue = this.queue.filter((item) => item.id !== id);
    this.notify();
  }

  public bulkResume(ids: string[]) {
    this.queue = this.queue.map((item) => {
      if (ids.includes(item.id) && (item.status === 'paused' || item.status === 'error')) {
        if (!item.videoInfo) {
          this.fetchInfoImmediately(item.id, item.url);
          return {
            ...item,
            status: 'fetching_info',
            statusText: 'Fetching details...',
            errorMessage: undefined,
          };
        }
        return {
          ...item,
          status: 'pending',
          statusText: 'Queued',
          errorMessage: undefined,
        };
      }
      return item;
    });
    this.notify();
    setTimeout(() => this.processNext(), 100);
  }

  public bulkRemove(ids: string[]) {
    ids.forEach((id) => {
      const controller = this.abortControllers.get(id);
      if (controller) {
        controller.abort();
        this.abortControllers.delete(id);
      }
    });
    this.queue = this.queue.filter((item) => !ids.includes(item.id));
    this.notify();
  }
}

export const queueService = new QueueService();

import * as FileSystem from 'expo-file-system/legacy';
import AsyncStorage from '@react-native-async-storage/async-storage';
import type { Track } from '../types';
import { formatDuration } from '../utils/audio';

const TRACKS_METADATA_KEY = 'tunes_native_tracks_meta';
const FAVORITES_KEY = 'tunes_favorites';
const TRACKS_DIR = `${FileSystem.documentDirectory}tracks/`;

// Ensure tracks directory exists on physical disk
const ensureDirectoryExists = async (): Promise<void> => {
  try {
    const dirInfo = await FileSystem.getInfoAsync(TRACKS_DIR);
    if (!dirInfo.exists) {
      await FileSystem.makeDirectoryAsync(TRACKS_DIR, { intermediates: true });
    }
  } catch (error) {
    console.error('[Storage] Error ensuring directory exists:', error);
  }
};

export const getTrackFilePath = (trackId: string): string => {
  return `${TRACKS_DIR}${trackId}.mp3`;
};

export const getAllTracks = async (): Promise<Track[]> => {
  try {
    const [rawTracks, rawFavs] = await Promise.all([
      AsyncStorage.getItem(TRACKS_METADATA_KEY),
      AsyncStorage.getItem(FAVORITES_KEY),
    ]);

    if (!rawTracks) return [];

    const list: Track[] = JSON.parse(rawTracks);
    const favSet = new Set<string>(rawFavs ? JSON.parse(rawFavs) : []);

    return list
      .map((track) => {
        // Self-heal previously saved tracks that have '--:--' or missing duration formatting
        let formatted = track.durationFormatted;
        if ((!formatted || formatted === '--:--') && track.duration && track.duration > 0) {
          formatted = formatDuration(track.duration);
        }

        return {
          ...track,
          durationFormatted: formatted || '--:--',
          isFavorite: favSet.has(track.id),
        };
      })
      .sort((a, b) => b.createdAt - a.createdAt);
  } catch (err) {
    console.error('[Storage] Failed to fetch tracks:', err);
    return [];
  }
};

export const saveTrack = async (track: Track): Promise<void> => {
  try {
    await ensureDirectoryExists();
    const existing = await getAllTracks();
    const updated = [track, ...existing.filter((t) => t.id !== track.id)];
    await AsyncStorage.setItem(TRACKS_METADATA_KEY, JSON.stringify(updated));
  } catch (err) {
    console.error('[Storage] Failed to save track:', err);
    throw err;
  }
};

export const toggleTrackFavorite = async (trackId: string): Promise<boolean> => {
  try {
    const rawFavs = await AsyncStorage.getItem(FAVORITES_KEY);
    const favs: string[] = rawFavs ? JSON.parse(rawFavs) : [];
    const isFav = favs.includes(trackId);
    const updated = isFav ? favs.filter((id) => id !== trackId) : [...favs, trackId];

    await AsyncStorage.setItem(FAVORITES_KEY, JSON.stringify(updated));
    return !isFav;
  } catch (err) {
    console.error('[Storage] Failed to toggle favorite:', err);
    return false;
  }
};

export const deleteTrack = async (id: string): Promise<void> => {
  try {
    const existing = await getAllTracks();
    const trackToDelete = existing.find((t) => t.id === id);

    if (trackToDelete && trackToDelete.fileUri) {
      const fileInfo = await FileSystem.getInfoAsync(trackToDelete.fileUri);
      if (fileInfo.exists) {
        await FileSystem.deleteAsync(trackToDelete.fileUri, { idempotent: true });
      }
    }

    const remaining = existing.filter((t) => t.id !== id);
    await AsyncStorage.setItem(TRACKS_METADATA_KEY, JSON.stringify(remaining));

    const rawFavs = await AsyncStorage.getItem(FAVORITES_KEY);
    if (rawFavs) {
      const favs: string[] = JSON.parse(rawFavs);
      await AsyncStorage.setItem(
        FAVORITES_KEY,
        JSON.stringify(favs.filter((itemId) => itemId !== id))
      );
    }
  } catch (err) {
    console.error('[Storage] Failed to delete track:', err);
  }
};

export const getStorageStats = async (): Promise<{ totalBytes: number; formattedSize: string }> => {
  try {
    await ensureDirectoryExists();
    const files = await FileSystem.readDirectoryAsync(TRACKS_DIR);
    let totalBytes = 0;

    for (const fileName of files) {
      const fileInfo = await FileSystem.getInfoAsync(`${TRACKS_DIR}${fileName}`);
      if (fileInfo.exists && !fileInfo.isDirectory) {
        totalBytes += fileInfo.size || 0;
      }
    }

    const formattedSize = `${(totalBytes / (1024 * 1024)).toFixed(1)} MB`;
    return { totalBytes, formattedSize };
  } catch {
    return { totalBytes: 0, formattedSize: '0.0 MB' };
  }
};

export const pruneStaleStorage = async (): Promise<{ freedBytes: number; prunedCount: number }> => {
  try {
    await ensureDirectoryExists();
    const tracks = await getAllTracks();
    const validFilePaths = new Set(tracks.map((t) => t.fileUri));

    const files = await FileSystem.readDirectoryAsync(TRACKS_DIR);
    let freedBytes = 0;
    let prunedCount = 0;

    for (const file of files) {
      const fullPath = `${TRACKS_DIR}${file}`;
      if (!validFilePaths.has(fullPath)) {
        const info = await FileSystem.getInfoAsync(fullPath);
        if (info.exists) {
          freedBytes += info.size || 0;
          await FileSystem.deleteAsync(fullPath, { idempotent: true });
          prunedCount++;
        }
      }
    }

    return { freedBytes, prunedCount };
  } catch {
    return { freedBytes: 0, prunedCount: 0 };
  }
};

export const updateTrackDuration = async (
  trackId: string,
  durationSeconds: number
): Promise<void> => {
  try {
    if (!trackId || !durationSeconds || durationSeconds <= 0) return;

    const raw = await AsyncStorage.getItem(TRACKS_METADATA_KEY);
    if (!raw) return;

    const tracks: Track[] = JSON.parse(raw);
    let hasChanged = false;

    const totalSecs = Math.floor(durationSeconds);
    const mins = Math.floor(totalSecs / 60);
    const secs = totalSecs % 60;
    const formatted = `${mins}:${secs.toString().padStart(2, '0')}`;

    const updated = tracks.map((t) => {
      if (t.id === trackId) {
        if (!t.duration || t.duration <= 0 || !t.durationFormatted || t.durationFormatted === '--:--') {
          hasChanged = true;
          return {
            ...t,
            duration: totalSecs,
            durationFormatted: formatted,
          };
        }
      }
      return t;
    });

    if (hasChanged) {
      await AsyncStorage.setItem(TRACKS_METADATA_KEY, JSON.stringify(updated));
    }
  } catch (err) {
    console.warn('[Storage] Failed to persist detected track duration:', err);
  }
};

export const findDuplicateTrack = async (preview: {
  id?: string;
  title?: string;
  channel?: string;
  thumbnail?: string;
}): Promise<Track | null> => {
  try {
    const tracks = await getAllTracks();
    const cleanTitle = (preview.title || '').trim().toLowerCase();
    const cleanChannel = (preview.channel || '').trim().toLowerCase();

    for (const t of tracks) {
      // 1. Direct ID match
      if (preview.id && t.id === preview.id) {
        return t;
      }
      // 2. Direct Thumbnail match (matches exact YouTube video asset)
      if (preview.thumbnail && t.thumbnailUrl && t.thumbnailUrl === preview.thumbnail) {
        return t;
      }
      // 3. Title + Channel / Artist match (ignoring duration)
      if (cleanTitle && cleanChannel) {
        const tTitle = (t.title || '').trim().toLowerCase();
        const tArtist = (t.artist || '').trim().toLowerCase();
        if (tTitle === cleanTitle && tArtist === cleanChannel) {
          return t;
        }
      }
    }
    return null;
  } catch {
    return null;
  }
};

interface ThemeContextType {
  tokens: ThemeTokens;
  shapes: {
    xs: number;
    sm: number;
    md: number;
    lg: number;
    sheet: number;
    pill: number;
  };
  currentPalette: PaletteStyle;
  setPalette: (palette: PaletteStyle) => void;
}

const ThemeContext = createContext<ThemeContextType | null>(null);

const STORAGE_KEY = 'pixel_music_palette_style';

export const ThemeProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [currentPalette, setCurrentPalette] = useState<PaletteStyle>('TONAL_SPOT');

  useEffect(() => {
    AsyncStorage.getItem(STORAGE_KEY).then((saved) => {
      if (saved && (saved in PALETTES)) {
        setCurrentPalette(saved as PaletteStyle);
      }
    });
  }, []);

  const setPalette = (palette: PaletteStyle) => {
    setCurrentPalette(palette);
    AsyncStorage.setItem(STORAGE_KEY, palette);
  };

  const tokens = useMemo(() => PALETTES[currentPalette], [currentPalette]);

  const shapes = useMemo(
    () => ({
      xs: 4,
      sm: 8,
      md: 16,
      lg: 24,
      sheet: 28,
      pill: 999,
    }),
    []
  );

  return (
    <ThemeContext.Provider value={{ tokens, shapes, currentPalette, setPalette }}>
      {children}
    </ThemeContext.Provider>
  );
};

export const useTheme = () => {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return context;
};
